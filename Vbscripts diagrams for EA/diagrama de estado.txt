' ------------------------------------------------------------
' SUB PRINCIPAL
' ------------------------------------------------------------

Sub Main()

    Dim package
    Dim cuKey, cuDict, diagramName, estados, transiciones

    ' Obtener el Package seleccionado
    Set package = Repository.GetTreeSelectedPackage()

    If package Is Nothing Then
        Session.Output "ERROR: Selecciona un Package en el Project Browser."
        Exit Sub
    End If

    ' Contador para el resumen final
    Dim totalDiagramas
    totalDiagramas = 0

    ' Iterar sobre cada caso de uso
    For Each cuKey In dictCU.Keys
        Set cuDict = dictCU.Item(cuKey)
        diagramName = cuKey & " - " & cuDict.Item("nombre")
        estados = cuDict.Item("estados")
        transiciones = cuDict.Item("transiciones")

        ' Crear el diagrama
        If CrearDiagramaActividad(package, diagramName, estados, transiciones) Then
            totalDiagramas = totalDiagramas + 1
        End If
    Next

    ' Resumen final
    Session.Output " "
    Session.Output "============================================"
    Session.Output " TOTAL DIAGRAMAS CREADOS: " & totalDiagramas & " de " & dictCU.Count
    Session.Output "============================================"

End Sub


' ------------------------------------------------------------
' FUNCIÓN: CREAR DIAGRAMA DE ACTIVIDAD
' ------------------------------------------------------------

Function CrearDiagramaActividad(package, diagramName, estados, transiciones)

    On Error Resume Next

    Dim diagram
    Dim elementos
    Dim flujos
    Dim i, j
    Dim estadoActual, transicionActual
    Dim origen, destino, condicion
    Dim posX, posY, ancho, alto
    Dim elementoObj
    Dim conector

    ' Crear el diagrama
    Set diagram = package.Diagrams.AddNew(diagramName, "Activity")
    If Err.Number <> 0 Then
        Session.Output "ERROR al crear diagrama " & diagramName & ": " & Err.Description
        CrearDiagramaActividad = False
        Exit Function
    End If
    diagram.Update()

    ' Diccionario para almacenar los elementos creados (por nombre)
    Dim elementosMap
    Set elementosMap = CreateObject("Scripting.Dictionary")

    ' Crear nodo inicial (si existe "Inicio" en la lista de estados)
    Dim inicioExiste
    inicioExiste = False
    For i = LBound(estados) To UBound(estados)
        If estados(i) = "Inicio" Then
            inicioExiste = True
            Exit For
        End If
    Next

    If inicioExiste Then
        Set elementoObj = CrearNodoInicial(package)
        elementosMap.Add "Inicio", elementoObj
    End If

    ' Crear nodo final (si existe "Fin" en la lista de estados)
    Dim finExiste
    finExiste = False
    For i = LBound(estados) To UBound(estados)
        If estados(i) = "Fin" Then
            finExiste = True
            Exit For
        End If
    Next

    If finExiste Then
        Set elementoObj = CrearNodoFinal(package)
        elementosMap.Add "Fin", elementoObj
    End If

    ' Crear acciones para el resto de estados (excluyendo Inicio y Fin)
    For i = LBound(estados) To UBound(estados)
        estadoActual = estados(i)
        If estadoActual <> "Inicio" And estadoActual <> "Fin" Then
            Set elementoObj = CrearAccion(package, estadoActual)
            elementosMap.Add estadoActual, elementoObj
        End If
    Next

    ' Ahora agregar los elementos al diagrama con posiciones automáticas
    ' Usaremos una cuadrícula vertical: cada estado en una fila, separados 60px
    ' El primer elemento (Inicio) en la primera fila, el último (Fin) en la última
    Dim numEstados
    numEstados = UBound(estados) - LBound(estados) + 1

    For i = LBound(estados) To UBound(estados)
        estadoActual = estados(i)
        If elementosMap.Exists(estadoActual) Then
            Set elementoObj = elementosMap.Item(estadoActual)
            ' Posición: centrado horizontal, espaciado vertical
            posX = 200  ' Centrado (ancho total 400-500)
            posY = 50 + (i - LBound(estados)) * 70
            ancho = 150
            alto = 40
            AgregarElemento diagram, elementoObj, posX, posY, posX + ancho, posY + alto
        End If
    Next

    ' Crear flujos (conexiones) entre elementos
    For j = LBound(transiciones) To UBound(transiciones)
        transicionActual = transiciones(j)

        ' Extraer origen, destino y condicion del formato "origen -> destino|condicion"
        Dim partes
        Dim flechaPos, barraPos
        flechaPos = InStr(transicionActual, " -> ")
        If flechaPos > 0 Then
            origen = Trim(Left(transicionActual, flechaPos - 1))
            barraPos = InStr(transicionActual, "|")
            If barraPos > 0 Then
                destino = Trim(Mid(transicionActual, flechaPos + 4, barraPos - flechaPos - 4))
                condicion = Mid(transicionActual, barraPos + 1)
            Else
                destino = Trim(Mid(transicionActual, flechaPos + 4))
                condicion = ""
            End If

            ' Verificar que origen y destino existen en el mapa
            If elementosMap.Exists(origen) And elementosMap.Exists(destino) Then
                CrearFlujo elementosMap.Item(origen), elementosMap.Item(destino), condicion
            Else
                Session.Output "ADVERTENCIA: No se encontró origen o destino en " & diagramName & " para: " & transicionActual
            End If
        Else
            Session.Output "ADVERTENCIA: Formato inválido en " & diagramName & ": " & transicionActual
        End If
    Next

    ' Actualizar el diagrama
    diagram.Update()
    package.Update()

    CrearDiagramaActividad = True

    On Error GoTo 0

End Function


' ============================================================
' FUNCIONES AUXILIARES (sin cambios)
' ============================================================

Function CrearAccion(package, nombre)

    Dim elemento
    Set elemento = package.Elements.AddNew(nombre, "Action")
    elemento.Update()
    Set CrearAccion = elemento

End Function


Function CrearNodoInicial(package)

    Dim elemento
    Set elemento = package.Elements.AddNew("Inicio", "StateNode")
    elemento.Subtype = 100
    elemento.Update()
    Set CrearNodoInicial = elemento

End Function


Function CrearNodoFinal(package)

    Dim elemento
    Set elemento = package.Elements.AddNew("Fin", "StateNode")
    elemento.Subtype = 101
    elemento.Update()
    Set CrearNodoFinal = elemento

End Function


Sub AgregarElemento(diagram, elemento, izquierda, arriba, derecha, abajo)

    Dim objeto
    Set objeto = diagram.DiagramObjects.AddNew( _
        "l=" & izquierda & _
        ";r=" & derecha & _
        ";t=" & arriba & _
        ";b=" & abajo & ";", _
        "")
    objeto.ElementID = elemento.ElementID
    objeto.Update()

End Sub


Sub CrearFlujo(origen, destino, condicion)

    Dim conector
    Set conector = origen.Connectors.AddNew("", "ControlFlow")
    conector.SupplierID = destino.ElementID
    conector.Update()

    If condicion <> "" Then
        conector.TransitionGuard = condicion
        conector.Update()
    End If

End Sub
